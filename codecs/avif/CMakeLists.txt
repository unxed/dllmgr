cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(dllmgr_avif VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This is a Windows COM DLL; use the Windows CI workflow.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
option(DLLMGR_AVIF_FIXTURE_ONLY "Build the test encoder, never ship it in the DLL" OFF)
set(DLLMGR_LIBAVIF_SOURCE "" CACHE PATH "Exact pinned libavif checkout")
set(DLLMGR_AOM_SOURCE "" CACHE PATH "Exact pinned libaom checkout")
find_package(Git REQUIRED)
function(require_commit dir commit)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${dir}" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual STREQUAL commit)
    message(FATAL_ERROR "Wrong source revision in ${dir}: ${actual}; require ${commit}")
  endif()
endfunction()
require_commit("${DLLMGR_LIBAVIF_SOURCE}" c5240fc79fe5c2407e10afd35f5505ef6333ea49)
require_commit("${DLLMGR_AOM_SOURCE}" 03087864cf4bea6abb0d28f95cf7843511413d8f)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(AVIF_CODEC_AOM LOCAL CACHE STRING "" FORCE)
set(AVIF_CODEC_AOM_DECODE ON CACHE BOOL "" FORCE)
set(AVIF_CODEC_AOM_ENCODE ${DLLMGR_AVIF_FIXTURE_ONLY} CACHE BOOL "" FORCE)
set(AVIF_LIBYUV OFF CACHE STRING "" FORCE)
set(AVIF_LIBSHARPYUV OFF CACHE STRING "" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(AOM_TARGET_CPU generic CACHE STRING "" FORCE)
set(CONFIG_STATIC_MSVCRT 1 CACHE BOOL "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_LIBAOM "${DLLMGR_AOM_SOURCE}" CACHE PATH "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
add_subdirectory("${DLLMGR_LIBAVIF_SOURCE}" libavif EXCLUDE_FROM_ALL)
if(DLLMGR_AVIF_FIXTURE_ONLY)
  add_executable(avif-fixtures fixtures.c)
  target_link_libraries(avif-fixtures PRIVATE avif)
else()
  add_library(dllmgr-avif SHARED codec.cpp codec.def)
  target_compile_definitions(dllmgr-avif PRIVATE _WIN32_WINNT=0x0601)
  target_link_libraries(dllmgr-avif PRIVATE avif windowscodecs ole32 uuid)
  set_target_properties(dllmgr-avif PROPERTIES PREFIX "")
  add_executable(avif-codec ../../tests/wine/avif-codec.c)
  target_link_libraries(avif-codec PRIVATE windowscodecs ole32 uuid advapi32 shlwapi)
  target_compile_definitions(avif-codec PRIVATE _WIN32_WINNT=0x0601)
  install(TARGETS dllmgr-avif avif-codec RUNTIME DESTINATION .)
  install(FILES README.md NOTICE.md codec_ids.h DESTINATION .)
  install(DIRECTORY licenses/ DESTINATION licenses)
  # Preserve bundled third-party notices as well as the two top-level licenses.
  file(GLOB_RECURSE aom_notices RELATIVE "${DLLMGR_AOM_SOURCE}" "${DLLMGR_AOM_SOURCE}/third_party/*LICENSE*" "${DLLMGR_AOM_SOURCE}/third_party/*PATENTS*" "${DLLMGR_AOM_SOURCE}/third_party/*COPYING*")
  foreach(notice IN LISTS aom_notices)
    get_filename_component(parent "${notice}" DIRECTORY)
    install(FILES "${DLLMGR_AOM_SOURCE}/${notice}" DESTINATION "licenses/aom/${parent}")
  endforeach()
endif()
